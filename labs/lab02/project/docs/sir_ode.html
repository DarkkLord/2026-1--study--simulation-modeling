function sir_ode!(du, u, p, t) (S, I, R) = u (β, c, γ) = p N = S + I + R @inbounds begin du[1] = -β * c * I / N * S du[2] = β * c * I / N * S - γ * I du[3] = γ * I end nothing end


Параметры модели

````@example sir_ode
δt = 0.1
tmax = 40.0
tspan = (0.0, tmax)
u0 = [990.0, 10.0, 0.0]  # S, I, R
p = [0.05, 10.0, 0.25]   # β, c, γ

Расчет базового репродуктивного числа

@example sir_ode R0 = (p[2] * p[1]) / p[3] # R₀ = (c * β) / γ

Создание и решение задачи

@example sir_ode prob_ode = ODEProblem(sir_ode!, u0, tspan, p) sol_ode = solve(prob_ode, dt = δt)

Подготовка данных в DataFrame

@example sir_ode df_ode = DataFrame(Tables.table(sol_ode')) rename!(df_ode, ["S", "I", "R"]) df_ode[!, :t] = sol_ode.t df_ode[!, :N] = df_ode.S + df_ode.I + df_ode.R # Общая численность популяции

Анализ модели для набора параметров

````@example sir_ode
parameter_sets = [
    [0.05, 10.0, 0.25],   # Базовый вариант
    [0.03, 10.0, 0.25],   # Меньшая вероятность заражения
    [0.07, 10.0, 0.25],   # Большая вероятность заражения
    [0.05, 10.0, 0.15],   # Более медленное выздоровление
    [0.05, 10.0, 0.35]    # Более быстрое выздоровление
]

println("\n=== РАСЧЁТ ДЛЯ НАБОРА ПАРАМЕТРОВ ===")

parameter_results = DataFrame(
    β = Float64[],
    c = Float64[],
    γ = Float64[],
    R0 = Float64[],
    I_max = Float64[],
    t_peak = Float64[],
    R_final = Float64[]
)

for params in parameter_sets
    β, c, γ = params

    R0_test = (c * β) / γ

    prob_test = ODEProblem(
        sir_ode!,
        u0,
        tspan,
        params
    )

    sol_test = solve(prob_test, dt=δt)

    I_test = [u[2] for u in sol_test.u]
    R_test = [u[3] for u in sol_test.u]

    peak_idx_test = argmax(I_test)
    peak_value_test = I_test[peak_idx_test]
    peak_time_test = sol_test.t[peak_idx_test]

    push!(
        parameter_results,
        (
            β,
            c,
            γ,
            R0_test,
            peak_value_test,
            peak_time_test,
            R_test[end]
        )
    )
end

println(parameter_results)
